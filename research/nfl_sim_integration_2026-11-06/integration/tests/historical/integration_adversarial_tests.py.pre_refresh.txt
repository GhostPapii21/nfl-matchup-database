def diagnostic_field_mapping(rows):
    """Expose every discovered wire difference; preserve raw producer separately.

    No score, seed, opportunity, identity, forecast/hash or RNG draw is created.
    Known unplayed OT zero rows are omitted ONLY in this diagnostic view because
    the market representation forbids them. Unmodified producer stays retained.
    """
    mapped = deepcopy(rows)
    for row in mapped:
        status = row["status"]
        row["producer_status_opaque"] = deepcopy(status)
        row.update(status=status["state"], completion_reason=status["completion_reason"],
                   terminal_state=deepcopy(status["terminal_state"]), went_ot=status["went_ot"])
        row["freeze"]["input_manifest_hash"] = row["freeze"]["input_manifest_digest"]
        row["engine"]["component_versions"] = {"joint_producer": row["engine"]["component_version"]}
        row["engine"]["dependency_versions"] = deepcopy(row["engine"]["dependencies"])
        row["team_periods"] = [p for p in row["team_periods"]
                               if status["went_ot"] or not p["period_id"].startswith("OT")]
        for period in row["team_periods"]:
            period["points"] = period["stats"]["points"]
    return mapped


def diagnostic_transport(rows, *, expected_trial_ids):
    """Test procedure requires independently retained ordered population explicitly."""
    if expected_trial_ids is None:
        raise ValueError("independently retained expected_trial_ids required by procedure")
    return adapter.adapt_joint_trials(rows, expected_trial_ids=expected_trial_ids)

@pytest.mark.parametrize("period,primitives", [
    ("H1", ("Q1", "Q2")), ("H2", ("Q3", "Q4")),
    ("REG", ("Q1", "Q2", "Q3", "Q4")),
    ("FG", ("Q1", "Q2", "Q3", "Q4", "OT1")),
])
def test_diagnostic_same_trial_score_primitives(producer_rows, period, primitives):
    trials = diagnostic_transport(diagnostic_field_mapping(producer_rows), expected_trial_ids=RETAINED_IDS)
    for raw, trial in zip(producer_rows, trials, strict=True):
        teams = (raw["game"]["home_team_id"], raw["game"]["away_team_id"])
        assert all(p["period_id"] in primitives for p in raw["team_periods"] if period == "FG")
        expected = tuple(sum(p["stats"]["points"] for p in raw["team_periods"]
                             if p["team_id"] == team and p["period_id"] in primitives) for team in teams)
        event_scores = tuple(sum(event["score_delta"][team] for event in raw["events"]
                                 if event["period_id"] in primitives) for team in teams)
        aggregate = tuple(next(p["stats"]["points"] for p in raw["aggregates"]["team"]
                               if p["team_id"] == team and p["period_id"] == period) for team in teams)
        assert trial.scorecard.points(period) == expected == event_scores == aggregate

        rows[0]["freeze"]["input_content_hash"] = "sha256:" + "0" * 64

        diagnostic_transport(rows, expected_trial_ids=RETAINED_IDS)

    elif missing == "phase15":
        p = root / "checklist.md"
        p.write_text(p.read_text().replace("| [x] | 15 |", "| [ ] | 15 |"))
    else:
        (root / missing).unlink()
    outcome = readiness.assess(manifest, root, allow_synthetic=True)
    assert outcome["projection_ready"] is False
    assert outcome["synthetic_fixture_ready"] is False and outcome["errors"]


def test_synthetic_gate_cannot_be_promoted_by_live_label(gate_fixture):
    manifest, root = gate_fixture
    manifest["synthetic"] = False
    outcome = readiness.assess(manifest, root)
    assert outcome["projection_ready"] is False
    assert any("synthetic" in reason.lower() for reason in outcome["errors"])

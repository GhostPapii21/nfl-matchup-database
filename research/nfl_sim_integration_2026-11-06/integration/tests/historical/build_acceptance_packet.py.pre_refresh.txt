def hash_file(path):
    return sha256(Path(path).read_bytes()).hexdigest()


def inventory(root):
    return [{'path': str(p), 'sha256': hash_file(p), 'bytes': p.stat().st_size}
            for p in sorted(root.rglob('*')) if p.is_file()
            and '__pycache__' not in p.parts and '.pytest_cache' not in p.parts]


def git(root, *args):
    command = ['git', '-C', str(root), *args]
    result = subprocess.run(command, text=True, capture_output=True)
    return dict(command=command, exit_code=result.returncode,
                stdout=result.stdout.strip(), stderr=result.stderr.strip())


def comparison(before, after):
    b, a = ({r['path']: r['sha256'] for r in rows} for rows in (before, after))
    return dict(changed=[p for p in b.keys() & a.keys() if b[p] != a[p]],
                added=sorted(a.keys() - b.keys()), removed=sorted(b.keys() - a.keys()),
                before_count=len(b), after_count=len(a), unchanged=b == a)


def write_json(path, value):
    path.write_text(json.dumps(value, indent=2) + '\n')

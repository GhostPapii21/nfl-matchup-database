"""Dated forward-only QB/injury adjustment layer — no fabricated histories."""
from __future__ import annotations
from typing import Any
import pandas as pd


def apply_forward_qb_filter(df: pd.DataFrame, adjustments: dict[str, Any]) -> tuple[pd.DataFrame, dict]:
    """Filter pass plays for forward sims. Does not mutate historical meaning of W1–8 rows used as catalog.
    When enabled, BAL pass catalog becomes starter-only (Fagnano); Lamar excluded; Huntley excluded from starter pool.
    Fabrication forbidden.
    """
    meta: dict[str, Any] = {}
    if not adjustments or not adjustments.get("enabled"):
        return df.copy(), {"enabled": False}
    out = df.copy()
    qb = adjustments.get("qb", {})
    for team, spec in qb.items():
        team_pass = out[(out["posteam"] == team) & (out["play_type"] == "pass")]
        n_before = int(len(team_pass))
        out_passers = list(spec.get("out") or [])
        doubtful = list(spec.get("doubtful_excluded_from_starter_pool") or [])
        starter = spec.get("starter")
        filt = spec.get("pass_play_filter")
        fabricated = bool(spec.get("fabricate_replacement_history", False))
        if fabricated:
            raise RuntimeError("fabricate_replacement_history must be false")
        excluded = out_passers + doubtful
        if filt == "starter_only_when_adjustments_enabled" and starter:
            # keep non-pass rows; for pass rows keep only starter passer
            keep_mask = ~((out["posteam"] == team) & (out["play_type"] == "pass")) | (
                out["passer_player_name"] == starter
            )
            out = out[keep_mask].copy()
        elif excluded:
            keep_mask = ~((out["posteam"] == team) & (out["play_type"] == "pass") & out["passer_player_name"].isin(excluded))
            out = out[keep_mask].copy()
        team_pass_after = out[(out["posteam"] == team) & (out["play_type"] == "pass")]
        meta[team] = {
            "starter": starter,
            "excluded_passers": out_passers,
            "doubtful_excluded": doubtful,
            "n_pass_before": n_before,
            "n_pass_after": int(len(team_pass_after)),
            "fabricated": False,
            "forward_only": True,
            "asof_et": adjustments.get("asof_et"),
            "sparse_qb": int(len(team_pass_after)) < 50,
        }
    meta["enabled"] = True
    meta["do_not_apply_retroactively_to_w1_8_history_rows"] = adjustments.get(
        "do_not_apply_retroactively_to_w1_8_history_rows", True
    )
    return out, meta

def load_config(path: str | Path) -> dict[str, Any]:
    p = Path(path)
    cfg = json.loads(p.read_text())
    required = ["config_id", "mode", "game", "history", "min_n_state", "proposed_pooling", "adjustments", "seed_base"]
    for k in required:
        if k not in cfg:
            raise ValueError(f"config missing {k}")
    if cfg["history"].get("no_league_fallback") is not True:
        raise ValueError("no_league_fallback must be true")
    return cfg


===== FILE: empirical.py =====
"""Empirical state → play lookup with explicit sparse statuses (no silent pool)."""
from __future__ import annotations
from dataclasses import dataclass
from typing import Optional
import numpy as np
import pandas as pd


def dist_bucket(ydstogo: float) -> str:
    y = float(ydstogo) if ydstogo == ydstogo else 10.0
    if y <= 3:
        return "short"
    if y <= 7:
        return "med"
    if y <= 10:
        return "long"
    return "xlong"


def yl_bucket(yardline_100: float) -> str:
    y = float(yardline_100) if yardline_100 == yardline_100 else 75.0
    if y <= 20:
        return "RZ"
    if y <= 50:
        return "opp50to20"
    if y <= 80:
        return "own20to50"
    return "backed"


@dataclass
class StateKey:
    posteam: str
    down: int
    dist_bucket: str
    yl_bucket: str


@dataclass
class Evidence:
    status: str  # OK | SPARSE_STATE | EMPTY_STATE
    n_evidence: int
    plays: Optional[pd.DataFrame]
    pooling_applied: bool = False
    pool_proposal: Optional[str] = None
    key: Optional[StateKey] = None


def _snap_plays(df: pd.DataFrame) -> pd.DataFrame:
    return df[df["play_type"].isin(["pass", "run", "qb_kneel", "qb_spike"])].copy()


def lookup_plays(
    df: pd.DataFrame,
    key: StateKey,
    *,
    min_n: int = 5,
    allow_pool: bool = False,
    proposed_pool_teams: Optional[list[str]] = None,
) -> Evidence:
    snaps = _snap_plays(df)
    snaps = snaps[snaps["posteam"] == key.posteam]
    snaps = snaps[snaps["down"] == key.down]
    if snaps.empty:
        return Evidence("EMPTY_STATE", 0, None, False, "PROPOSED_coarser_down_or_team_pool_NOT_APPLIED", key)
    # annotate buckets
    d = snaps.copy()
    d["_db"] = d["ydstogo"].map(dist_bucket)
    d["_yl"] = d["yardline_100"].map(yl_bucket)
    cell = d[(d["_db"] == key.dist_bucket) & (d["_yl"] == key.yl_bucket)]
    n = int(len(cell))
    if n >= min_n:
        return Evidence("OK", n, cell.drop(columns=["_db", "_yl"]), False, None, key)
    # sparse — never silent pool
    proposal = (
        f"PROPOSED_coarser_bins_or_same_team_down_pool n={n}<{min_n}; "
        f"allow_pool={allow_pool} proposed_teams={proposed_pool_teams}"
    )
    if allow_pool and proposed_pool_teams:
        # still require explicit allow_pool True from config — vertical slice documents this path
        pooled = d  # same team all dist/yl for this down — still team-only
        if len(pooled) >= min_n:
            return Evidence(
                "OK_POOLED_EXPLICIT",
                int(len(pooled)),
                pooled.drop(columns=["_db", "_yl"]),
                True,
                proposal,
                key,
            )
    return Evidence("SPARSE_STATE", n, cell.drop(columns=["_db", "_yl"]) if n else None, False, proposal, key)


def sample_play(rng: np.random.Generator, ev: Evidence) -> tuple[Optional[pd.Series], Evidence]:
    if ev.status not in {"OK", "OK_POOLED_EXPLICIT"} or ev.plays is None or len(ev.plays) == 0:
        return None, ev
    idx = int(rng.integers(0, len(ev.plays)))
    return ev.plays.iloc[idx], ev


def lookup_special(
    df: pd.DataFrame,
    posteam: str,
    kind: str,
) -> Evidence:
    """kind: punt | field_goal | extra_point | kickoff"""
    if kind == "punt":
        sub = df[(df["posteam"] == posteam) & (df["play_type"] == "punt")]
    elif kind == "field_goal":
        sub = df[(df["posteam"] == posteam) & (df["play_type"] == "field_goal")]
    elif kind == "extra_point":
        sub = df[(df["posteam"] == posteam) & (df["play_type"] == "extra_point")]
    elif kind == "kickoff":
        sub = df[df["play_type"] == "kickoff"]
    else:
        raise ValueError(kind)
    n = int(len(sub))
    if n == 0:
        return Evidence("EMPTY_STATE", 0, None, False, f"PROPOSED_league_{kind}_NOT_APPLIED")
    status = "OK" if n >= 3 else "SPARSE_STATE"
    return Evidence(status if status == "OK" else "SPARSE_STATE", n, sub if n >= 3 else (sub if n else None), False,
                    None if n >= 3 else f"PROPOSED_league_{kind}_NOT_APPLIED n={n}")

"""Load team-only history — never silent league/future fallback."""
from __future__ import annotations
from pathlib import Path
from typing import Sequence
import pandas as pd

COLS = [
    "game_id", "season", "week", "home_team", "away_team", "posteam", "defteam",
    "qtr", "down", "ydstogo", "yardline_100", "game_seconds_remaining",
    "play_type", "yards_gained", "touchdown", "interception", "fumble_lost",
    "field_goal_result", "extra_point_result", "two_point_conv_result",
    "kickoff_attempt", "punt_attempt", "passer_player_name", "rusher_player_name",
    "receiver_player_name", "complete_pass", "incomplete_pass", "sack",
    "first_down", "fixed_drive_result", "drive", "play_id", "desc",
    "total_home_score", "total_away_score", "epa", "air_yards",
    "qb_scramble",
    "drive_inside20",
    "pass_location",
    "qb_hit",
    "third_down_failed",
    "third_down_converted",
]


def load_team_history(
    pbp_path: str | Path,
    *,
    teams: Sequence[str],
    season: int,
    weeks: tuple[int, int],
) -> dict:
    path = Path(pbp_path)
    if not path.exists():
        return {"status": "PBP_MISSING", "df": None, "evidence": {}}
    import pyarrow.parquet as pq
    available = set(pq.ParquetFile(path).schema_arrow.names)
    use = [c for c in COLS if c in available]
    df = pd.read_parquet(path, columns=use)
    w0, w1 = weeks
    # Games involving either team, weeks in range, season match
    invol = df["home_team"].isin(teams) | df["away_team"].isin(teams)
    mask = invol & (df["season"] == season) & (df["week"] >= w0) & (df["week"] <= w1)
    games = df.loc[mask].copy()
    # Offense rows: posteam in teams (team-only offense catalog)
    off = games[games["posteam"].isin(teams)].copy()
    if off.empty:
        return {"status": "EMPTY_TEAM_HISTORY", "df": off, "evidence": {"n_games": 0, "n_plays": 0}}
    # Guard: no future weeks
    if off["week"].max() > w1 or off["week"].min() < w0:
        return {"status": "WEEK_FILTER_FAILED", "df": off, "evidence": {}}
    return {
        "status": "OK",
        "df": off.reset_index(drop=True),
        "games_df": games.reset_index(drop=True),
        "evidence": {
            "n_games": int(games["game_id"].nunique()),
            "n_plays": int(len(off)),
            "teams": list(teams),
            "weeks": [w0, w1],
            "season": season,
            "source": str(path),
        },
    }


def load_involvement_plays(pbp_path, *, teams, season, weeks):
    """All plays from games involving teams (includes opponent offense vs our defense)."""
    import pyarrow.parquet as pq
    from pathlib import Path
    path = Path(pbp_path)
    available = set(pq.ParquetFile(path).schema_arrow.names)
    use = [c for c in COLS if c in available]
    df = __import__("pandas").read_parquet(path, columns=use)
    w0, w1 = weeks
    invol = df["home_team"].isin(teams) | df["away_team"].isin(teams)
    mask = invol & (df["season"] == season) & (df["week"] >= w0) & (df["week"] <= w1)
    return df.loc[mask].reset_index(drop=True)

"""Run bounded JAX@BAL possession sims under frozen configs; write artifacts. Research only."""
from __future__ import annotations
import json, time
from pathlib import Path
from config_load import load_config
from pbp_filter import load_team_history
from situational import build_situational_report
from provenance import build_default_registry, write_registry
from adjustment_ledger import build_ledger, write_ledger
from simulate import simulate_games, stable_seed
from adjustments import apply_forward_qb_filter

ROOT = Path(__file__).resolve().parent
PBP = ROOT.parent / "period_engine_v0/data/play_by_play_2026.parquet"
ART = ROOT / "artifacts"
ART.mkdir(exist_ok=True)


def main():
    hist = load_team_history(PBP, teams=["JAX", "BAL"], season=2026, weeks=(1, 8))
    sit = build_situational_report(hist["df"], hist.get("games_df"))
    (ART / "situational_jax_bal_w1_8.json").write_text(json.dumps(sit, indent=2) + "\n")

    results = {}
    # Practical N: try 10k; if slow report — target <120s for 10k
    # Practical N: ~20ms/game → 2k≈40s/config. 100k≈33min/config = RUNTIME_BLOCKER this session.
    PRACTICAL_N = 2000
    for cfg_name, n in [
        ("jax_bal_tnf_2026-11-05_data_only.json", PRACTICAL_N),
        ("jax_bal_tnf_2026-11-05_script_adjusted.json", PRACTICAL_N),
    ]:
        cfg = load_config(ROOT / "config" / cfg_name)
        seed = stable_seed(cfg["seed_base"], 9, cfg["config_id"])
        t0 = time.perf_counter()
        out = simulate_games(cfg, pbp_path=PBP, n=n, seed=seed)
        elapsed = time.perf_counter() - t0
        # strip large arrays to summary
        import numpy as np
        A = out["arrays"]
        summary = {
            k: v for k, v in out.items() if k not in ("arrays", "player_joint")
        }
        summary["runtime_sec"] = elapsed
        summary["score_means"] = {
            "fg_home": float(np.mean(A["fg_home"])),
            "fg_away": float(np.mean(A["fg_away"])),
            "reg_home": float(np.mean(A["reg_home"])),
            "reg_away": float(np.mean(A["reg_away"])),
            "p_ot": float(np.mean(A["went_ot"])),
        }
        summary["player_joint_top"] = {
            kind: {tm: dict(sorted(pl.items(), key=lambda x: -x[1])[:5]) for tm, pl in teams.items()}
            for kind, teams in out["player_joint"].items()
        }
        results[cfg["config_id"]] = summary
        write_ledger(ART / f"ledger_{cfg['config_id']}.json", build_ledger(cfg))
        print(cfg["config_id"], "n", out["n_completed"], "sec", round(elapsed, 2), "status", out["status"], "means", summary["score_means"])

    # provenance with script_adjusted meta
    cfg_adj = load_config(ROOT / "config/jax_bal_tnf_2026-11-05_script_adjusted.json")
    _, adj_meta = apply_forward_qb_filter(hist["df"], cfg_adj["adjustments"])
    reg = build_default_registry(hist_evidence=hist["evidence"], adj_meta=adj_meta, situational_ns=sit["ns"])
    write_registry(ART / "input_provenance_registry.json", reg)

    # market informed: document only (sampler not conditioned yet)
    cfg_m = load_config(ROOT / "config/jax_bal_tnf_2026-11-05_market_informed.json")
    write_ledger(ART / f"ledger_{cfg_m['config_id']}.json", build_ledger(cfg_m))
    results["market_informed_note"] = {
        "config_id": cfg_m["config_id"],
        "sampler_conditioned": False,
        "quotes_status": cfg_m["market_conditioning"]["quotes_status"],
        "contaminates_data_only": False,
        "note": "Config frozen separate; conditioning not wired to sampler — avoids false baseline contamination",
    }

    results["run_size_policy"] = {
        "practical_n": PRACTICAL_N,
        "per_game_ms_approx": 20,
        "n_100k_status": "RUNTIME_BLOCKER",
        "n_100k_est_sec_per_config": 2000,
        "note": "Do not fabricate 100k outputs; re-run when compute budget allows",
        "score_level_limitation": "Vertical slice empirical offense bootstrap overstates mean points vs NFL norms — not calibrated; research only",
    }
    (ART / "sim_jaxbal_summary.json").write_text(json.dumps(results, indent=2) + "\n")
    print("wrote artifacts to", ART)


if __name__ == "__main__":
    main()

"""Possession/PBP Monte Carlo vertical slice — team-only empirical resampling."""
from __future__ import annotations
from pathlib import Path
from typing import Any, Optional
import hashlib
import numpy as np
import pandas as pd

from pbp_filter import load_team_history
from empirical import StateKey, lookup_plays, lookup_special, sample_play, dist_bucket, yl_bucket

def stable_seed(base: int, week: int, game_id: str) -> int:
    digest = hashlib.sha256(f"{base}|{week}|{game_id}".encode()).hexdigest()
    return (base + week * 1000 + int(digest[:8], 16)) % (2**31 - 1)


def _clock_cost(play_type: str, rng: np.random.Generator) -> int:
    means = {"pass": 22, "run": 28, "punt": 10, "field_goal": 7, "extra_point": 4, "qb_kneel": 40, "qb_spike": 2}
    mu = means.get(play_type, 24)
    return int(max(3, rng.normal(mu, 4)))


def _credit_player(st: GameState, team: str, kind: str, name: Optional[str], yards: float) -> None:
    if not isinstance(name, str) or not name or name == "nan":
        return
    bucket = st.player_counts.setdefault(kind, {}).setdefault(team, {})
    bucket[name] = bucket.get(name, 0.0) + float(yards)


def _other(st: GameState, team: str) -> str:
    return st.away if team == st.home else st.home


class PlayCatalog:
    """Pre-index team snaps for fast sampling."""

    def __init__(self, df: pd.DataFrame, min_n: int):
        self.df = df
        self.min_n = min_n
        self.snaps = df[df["play_type"].isin(["pass", "run", "qb_kneel", "qb_spike"])].copy()
        self.snaps["_db"] = self.snaps["ydstogo"].map(dist_bucket)
        self.snaps["_yl"] = self.snaps["yardline_100"].map(yl_bucket)
        self.cells: dict[tuple, pd.DataFrame] = {}
        self.by_down: dict[tuple, pd.DataFrame] = {}
        self.by_team: dict[str, pd.DataFrame] = {}
        for team, g in self.snaps.groupby("posteam"):
            self.by_team[team] = g
            for down, gd in g.groupby("down"):
                self.by_down[(team, int(down))] = gd
                for (db, yl), cell in gd.groupby(["_db", "_yl"]):
                    self.cells[(team, int(down), str(db), str(yl))] = cell
        self.fourth_probs = {}
        self.specials = {"punt": {}, "field_goal": {}, "extra_point": {}}
        for team, g in df.groupby("posteam"):
            fourth = g[g["down"] == 4]
            if len(fourth) >= 5:
                types = fourth["play_type"].value_counts(normalize=True)
                p_punt = float(types.get("punt", 0)); p_fg = float(types.get("field_goal", 0))
                p_go = float(types.get("pass", 0) + types.get("run", 0)); s = p_punt + p_fg + p_go
                self.fourth_probs[team] = (p_punt/s, p_fg/s, p_go/s) if s > 0 else (1.0, 0.0, 0.0)
            for kind, ptype in [("punt","punt"),("field_goal","field_goal"),("extra_point","extra_point")]:
                sub = g[g["play_type"] == ptype]
                if len(sub) >= 3:
                    self.specials[kind][team] = sub.reset_index(drop=True)

    def sample_snap(self, rng: np.random.Generator, st: GameState, allow_pool: bool) -> tuple[Optional[pd.Series], str]:
        key = (st.possession, int(st.down), dist_bucket(st.ydstogo), yl_bucket(st.yardline_100))
        cell = self.cells.get(key)
        if cell is not None and len(cell) >= self.min_n:
            return cell.iloc[int(rng.integers(0, len(cell)))], "OK"
        # team-only coarser fallbacks with explicit flags (NOT league)
        down_key = (st.possession, int(st.down))
        broad = self.by_down.get(down_key)
        if broad is not None and len(broad) > 0:
            return broad.iloc[int(rng.integers(0, len(broad)))], f"SPARSE_STATE_FALLBACK_SAME_DOWN n_cell={0 if cell is None else len(cell)}"
        team = self.by_team.get(st.possession)
        if team is not None and len(team) > 0:
            return team.iloc[int(rng.integers(0, len(team)))], "SPARSE_STATE_FALLBACK_ANY_DOWN_SAME_TEAM"
        return None, "EMPTY_TEAM_SNAPS"


def _fourth_down_decision(rng: np.random.Generator, catalog: PlayCatalog, st: GameState) -> str:
    probs = catalog.fourth_probs.get(st.possession)
    if probs is None:
        st.sparse_flags.append("SPARSE_4TH_DECISION default_punt")
        return "punt"
    return str(rng.choice(["punt", "field_goal", "go"], p=list(probs)))


def _attempt_fg(rng: np.random.Generator, catalog: PlayCatalog, st: GameState) -> None:
    plays = catalog.specials["field_goal"].get(st.possession)
    st.game_seconds_remaining -= _clock_cost("field_goal", rng)
    if plays is None or len(plays) == 0:
        st.sparse_flags.append("SPARSE_FG")
        st.possession = _other(st, st.possession)
        st.yardline_100 = int(np.clip(100 - st.yardline_100, 20, 80))
        st.down, st.ydstogo = 1, 10
        st.events.append({"type": "fg_sparse_miss_default"})
        return
    row = plays.iloc[int(rng.integers(0, len(plays)))]
    res = str(row.get("field_goal_result") or "")
    if res == "made":
        st.add_score(st.possession, 3)
        st.events.append({"type": "field_goal", "result": "made", "team": st.possession})
        apply_kickoff(st, receiving=_other(st, st.possession), yardline_100=75)
        if st.in_ot:
            st.final = True
    else:
        st.events.append({"type": "field_goal", "result": res or "missed", "team": st.possession})
        st.possession = _other(st, st.possession)
        st.yardline_100 = int(np.clip(100 - st.yardline_100, 20, 80))
        st.down, st.ydstogo = 1, 10


def _attempt_pat(rng: np.random.Generator, catalog: PlayCatalog, st: GameState, scoring_team: str) -> None:
    plays = catalog.specials["extra_point"].get(scoring_team)
    st.game_seconds_remaining -= _clock_cost("extra_point", rng)
    if plays is None or len(plays) == 0:
        st.sparse_flags.append("SPARSE_PAT assume_made")
        st.add_score(scoring_team, 1)
        st.events.append({"type": "pat", "result": "assume_made_sparse", "team": scoring_team})
        return
    row = plays.iloc[int(rng.integers(0, len(plays)))]
    res = str(row.get("extra_point_result") or "")
    if res == "good":
        st.add_score(scoring_team, 1)
    st.events.append({"type": "pat", "result": res or "unknown", "team": scoring_team})


def _punt_fixed(rng: np.random.Generator, catalog: PlayCatalog, st: GameState) -> None:
    offense = st.possession
    yl = st.yardline_100
    plays = catalog.specials["punt"].get(offense)
    st.game_seconds_remaining -= _clock_cost("punt", rng)
    net = 40
    if plays is not None and len(plays):
        yg = pd.to_numeric(plays["yards_gained"], errors="coerce").dropna()
        if len(yg):
            net = int(np.clip(abs(float(yg.iloc[int(rng.integers(0, len(yg)))])), 25, 60))
    else:
        st.sparse_flags.append("SPARSE_PUNT default_net40")
    # Ball advances net yards toward offense's EZ; receiving team's yardline_100 flips.
    ball_yl_from_kick_opp = int(np.clip(yl - net, 1, 99))
    recv_yl = int(np.clip(100 - ball_yl_from_kick_opp, 10, 90))
    st.possession = _other(st, offense)
    st.yardline_100 = recv_yl
    st.down, st.ydstogo = 1, 10
    st.events.append({"type": "punt", "from_yl": yl, "net": net, "new_yl": recv_yl, "kick_ball_yl": ball_yl_from_kick_opp, "team": offense})


def _apply_snap(rng: np.random.Generator, catalog: PlayCatalog, st: GameState, allow_pool: bool) -> None:
    row, status = catalog.sample_snap(rng, st, allow_pool)
    if status != "OK":
        st.sparse_flags.append(status)
    if row is None:
        _punt_fixed(rng, catalog, st)
        return

    play_type = str(row["play_type"])
    yards = float(row["yards_gained"]) if pd.notna(row["yards_gained"]) else 0.0
    st.game_seconds_remaining -= _clock_cost(play_type, rng)

    if int(row.get("interception") or 0) == 1 or int(row.get("fumble_lost") or 0) == 1:
        st.events.append({"type": "turnover", "play_type": play_type, "team": st.possession})
        st.possession = _other(st, st.possession)
        st.yardline_100 = int(np.clip(100 - max(st.yardline_100 - yards, 1), 10, 90))
        st.down, st.ydstogo = 1, 10
        return

    if play_type == "pass":
        _credit_player(st, st.possession, "pass_yards", row.get("passer_player_name"), max(yards, 0))
        if int(row.get("complete_pass") or 0) == 1:
            _credit_player(st, st.possession, "rec_yards", row.get("receiver_player_name"), max(yards, 0))
    elif play_type == "run":
        _credit_player(st, st.possession, "rush_yards", row.get("rusher_player_name"), yards)

    # Score only if yards from *current* field position reach the EZ.
    # Do NOT trust historical `touchdown` flags from a different yardline context.
    if (st.yardline_100 - yards) <= 0:
        scoring = st.possession
        st.add_score(scoring, 6)
        st.events.append({"type": "touchdown", "team": scoring, "yards": yards})
        _attempt_pat(rng, catalog, st, scoring)
        apply_kickoff(st, receiving=_other(st, scoring), yardline_100=75)
        if st.in_ot:
            st.final = True
            st.events.append({"type": "final_ot"})
        return

    st.yardline_100 = int(np.clip(st.yardline_100 - yards, 1, 99))
    if yards >= st.ydstogo:
        st.down = 1
        st.ydstogo = 10 if st.yardline_100 >= 10 else max(1, st.yardline_100)
    else:
        st.down += 1
        st.ydstogo = max(1, int(st.ydstogo - yards))
        if st.down > 4:
            st.events.append({"type": "turnover_on_downs", "team": st.possession})
            st.possession = _other(st, st.possession)
            st.yardline_100 = int(np.clip(100 - st.yardline_100, 10, 90))
            st.down, st.ydstogo = 1, 10
            return
    st.events.append({"type": "snap", "play_type": play_type, "yards": yards, "down": st.down})


def _advance_period(st: GameState, home: str) -> None:
    if st.qtr == 1:
        st.qtr = 2
        st.game_seconds_remaining = 900
        st.events.append({"type": "end_q1"})
    elif st.qtr == 2:
        maybe_halftime(st, q3_receive=home)
    elif st.qtr == 3:
        st.qtr = 4
        st.game_seconds_remaining = 900
        st.events.append({"type": "end_q3"})
    elif st.qtr == 4:
        start_ot_if_tied(st)
    elif st.qtr >= 5:
        st.final = True
        st.events.append({"type": "final_ot_tie"})


def simulate_one(df: pd.DataFrame, catalog: PlayCatalog, cfg: dict, rng: np.random.Generator) -> GameState:
    home = cfg["game"]["home"]
    away = cfg["game"]["away"]
    st = GameState.new_game(home=home, away=away)
    st.game_seconds_remaining = 900  # Q1 only
    apply_kickoff(st, receiving=away, yardline_100=75)
    allow_pool = bool(cfg.get("proposed_pooling", {}).get("enabled", False))
    steps = 0
    max_steps = 220
    while not st.final and steps < max_steps:
        steps += 1
        if st.game_seconds_remaining <= 0:
            _advance_period(st, home)
            continue
        if st.down == 4:
            decision = _fourth_down_decision(rng, catalog, st)
            if decision == "punt":
                _punt_fixed(rng, catalog, st)
                continue
            if decision == "field_goal":
                _attempt_fg(rng, catalog, st)
                continue
        _apply_snap(rng, catalog, st, allow_pool=allow_pool)
        if st.in_ot and st.final:
            break
    if not st.final:
        if st.reg_score_home is None:
            st.reg_score_home, st.reg_score_away = st.score_home, st.score_away
        st.final = True
        st.events.append({"type": "final_max_steps" if steps >= max_steps else "final"})
    if st.reg_score_home is None:
        st.reg_score_home, st.reg_score_away = st.score_home, st.score_away
    return st


def simulate_games(cfg: dict, *, pbp_path: str | Path, n: int, seed: int) -> dict[str, Any]:
    hist = load_team_history(
        pbp_path,
        teams=cfg["history"]["teams"],
        season=cfg["history"]["season"],
        weeks=(cfg["history"]["weeks_inclusive"][0], cfg["history"]["weeks_inclusive"][-1]),
    )
    if hist["status"] != "OK":
        return {"status": hist["status"], "n_completed": 0, "error": "history_load_failed", "pooling_silently_applied": False}

    df = hist["df"]
    adj_meta: dict = {"enabled": False}
    qb_status: dict = {}
    if cfg.get("adjustments", {}).get("enabled"):
        df, adj_meta = apply_forward_qb_filter(df, cfg["adjustments"])
        for team, m in adj_meta.items():
            if isinstance(m, dict) and m.get("sparse_qb"):
                qb_status[team] = f"SPARSE_QB_HISTORY n_pass={m.get('n_pass_after')}"

    catalog = PlayCatalog(df, min_n=int(cfg["min_n_state"]))
    rng = np.random.default_rng(seed)
    buckets = {k: [] for k in [
        "q1_home","q1_away","q2_home","q2_away","q3_home","q3_away","q4_home","q4_away",
        "h1_home","h1_away","h2_home","h2_away","ot_home","ot_away",
        "reg_home","reg_away","fg_home","fg_away","went_ot",
    ]}
    joint_acc = {"pass_yards": {}, "rush_yards": {}, "rec_yards": {}}
    sparse_counts = 0
    for _ in range(n):
        st = simulate_one(df, catalog, cfg, rng)
        snap = st.period_snapshot()
        # Prefer explicit reg freeze when set
        if st.reg_score_home is not None:
            snap["reg_home"] = st.reg_score_home
            snap["reg_away"] = st.reg_score_away
        for k in buckets:
            if k == "went_ot":
                buckets[k].append(bool(st.in_ot))
            else:
                buckets[k].append(snap[k])
        if st.sparse_flags:
            sparse_counts += 1
        for kind in ("pass_yards", "rush_yards", "rec_yards"):
            for team, players in st.player_counts.get(kind, {}).items():
                bucket = joint_acc.setdefault(kind, {}).setdefault(team, {})
                for plyr, yds in players.items():
                    bucket[plyr] = bucket.get(plyr, 0.0) + yds

    for kind in joint_acc:
        for team in joint_acc[kind]:
            for plyr in list(joint_acc[kind][team]):
                joint_acc[kind][team][plyr] = joint_acc[kind][team][plyr] / n

    status = "OK_WITH_SPARSE_FLAGS" if sparse_counts else "OK"
    return {
        "status": status,
        "n_completed": n,
        "n_requested": n,
        "seed": seed,
        "config_id": cfg["config_id"],
        "mode": cfg["mode"],
        "pooling_silently_applied": False,
        "proposed_pooling_enabled": bool(cfg.get("proposed_pooling", {}).get("enabled", False)),
        "history_evidence": hist["evidence"],
        "adjustments_meta": adj_meta,
        "qb_status": qb_status,
        "evidence_summary": {
            "min_n_state": cfg["min_n_state"],
            "games_with_sparse_flags": sparse_counts,
            "sparse_rate": sparse_counts / n if n else None,
        },
        "arrays": {k: np.asarray(v) for k, v in buckets.items()},
        "player_joint": joint_acc,
        "limitations": [
            "Vertical slice: empirical team-only snap resampling + explicit same-team coarser fallbacks",
            "Clock costs are fixed means by play type — not a fitted clock model",
            "OT rule is single-possession approx (declared)",
            "No league silent pool; proposed pooling config default false",
            "W1-8 development set; no betting edge claim",
            "Inspired conceptually by NFLSimulatoR/nflfastR (MIT); no R source copied",
            "See FEATURE_MATRIX.md for unimplemented coverage/route/pressure fields",
        ],
        "source_composition": "team_only_2026_w1_8_JAX_BAL_empirical_not_possession_EP_model",
    }

"""Situational opportunity + conditional success tables (team-only W1–8)."""
from __future__ import annotations
from typing import Any
import numpy as np

def _rate(num: int, den: int) -> float | None:
    return float(num) / den if den else None


def third_down_table(df: pd.DataFrame) -> dict[str, Any]:
    out = {"by_team": {}, "n_total": 0}
    snaps = df[df["down"] == 3].copy()
    out["n_total"] = int(len(snaps))
    for team, g in snaps.groupby("posteam"):
        # prefer explicit flags when present
        if "third_down_converted" in g.columns:
            opp = g[g["play_type"].isin(["pass", "run", "qb_kneel", "qb_spike"])]
            conv = int((opp["third_down_converted"] == 1).sum())
            fail = int((opp["third_down_failed"] == 1).sum()) if "third_down_failed" in opp.columns else int(len(opp) - conv)
            den = conv + fail if (conv + fail) > 0 else int(len(opp))
        else:
            opp = g[g["play_type"].isin(["pass", "run"])]
            conv = int((opp["first_down"] == 1).sum())
            den = int(len(opp))
            fail = den - conv
        out["by_team"][team] = {
            "n_opportunities": den,
            "n_conversions": conv,
            "n_failures": fail,
            "p_convert_given_opportunity": _rate(conv, den),
            "status": "OK" if den >= 20 else "SPARSE",
        }
    return out


def red_zone_table(df: pd.DataFrame) -> dict[str, Any]:
    out = {"by_team": {}, "n_total": 0}
    rz = df[df["yardline_100"] <= 20].copy()
    rz = rz[rz["play_type"].isin(["pass", "run", "field_goal"])]
    out["n_total"] = int(len(rz))
    for team, g in rz.groupby("posteam"):
        # opportunity ≈ unique game+drive in RZ if drive col present else plays
        if "drive" in g.columns:
            trips = g.groupby(["game_id", "drive"]).agg(
                td=("touchdown", "max"),
                fg=("play_type", lambda s: int((s == "field_goal").any())),
            )
            n_trips = int(len(trips))
            n_td = int((trips["td"] == 1).sum())
            out["by_team"][team] = {
                "n_rz_trips": n_trips,
                "n_rz_td_drives": n_td,
                "p_td_given_rz_trip": _rate(n_td, n_trips),
                "n_rz_plays": int(len(g)),
                "status": "OK" if n_trips >= 8 else "SPARSE",
            }
        else:
            out["by_team"][team] = {"n_rz_plays": int(len(g)), "status": "SPARSE", "note": "no drive col"}
    return out


def qb_hit_completion(df: pd.DataFrame) -> dict[str, Any]:
    """PROXY: qb_hit split — NOT true pressure/blitz."""
    out = {"proxy_label": "qb_hit_NOT_measured_pressure_or_blitz", "by_passer": {}, "n_total": 0}
    passes = df[df["play_type"] == "pass"].copy()
    if "qb_hit" not in passes.columns:
        return {"status": "MISSING_COLUMN", "proxy_label": out["proxy_label"]}
    out["n_total"] = int(len(passes))
    for (team, qb), g in passes.groupby(["posteam", "passer_player_name"]):
        if not isinstance(qb, str):
            continue
        hit = g[g["qb_hit"] == 1]
        no = g[g["qb_hit"] != 1]
        def _comp(x):
            if "complete_pass" not in x.columns or len(x) == 0:
                return None, 0
            return _rate(int((x["complete_pass"] == 1).sum()), int(len(x))), int(len(x))
        c_hit, n_hit = _comp(hit)
        c_no, n_no = _comp(no)
        out["by_passer"][f"{team}:{qb}"] = {
            "n_hit": n_hit,
            "comp_given_hit": c_hit,
            "n_no_hit": n_no,
            "comp_given_no_hit": c_no,
            "status": "SPARSE" if (n_hit + n_no) < 20 else "PROXY_OK",
        }
    out["status"] = "PROXY"
    return out


def rb_ypc_vs_defense(df: pd.DataFrame, games_df: pd.DataFrame | None = None) -> dict[str, Any]:
    """Opponent-adjusted RB YPC: offense runs grouped by defteam (team-only window)."""
    runs = df[df["play_type"] == "run"].copy()
    out = {"by_offense_vs_def": {}, "status": "OK"}
    if runs.empty or "defteam" not in runs.columns:
        return {"status": "MISSING", "by_offense_vs_def": {}}
    for (off, deff), g in runs.groupby(["posteam", "defteam"]):
        ypc = float(pd.to_numeric(g["yards_gained"], errors="coerce").mean())
        out["by_offense_vs_def"][f"{off}_vs_{deff}"] = {
            "n_carries": int(len(g)),
            "ypc": ypc,
            "status": "OK" if len(g) >= 10 else "SPARSE",
        }
    # Highlight JAX vs BAL and BAL vs JAX if present
    out["matchup_jax_vs_bal"] = out["by_offense_vs_def"].get("JAX_vs_BAL", {"status": "SPARSE", "n_carries": 0, "note": "no W1–8 H2H in window"})
    out["matchup_bal_vs_jax"] = out["by_offense_vs_def"].get("BAL_vs_JAX", {"status": "SPARSE", "n_carries": 0, "note": "no W1–8 H2H in window"})
    return out


def pass_location_counts(df: pd.DataFrame) -> dict[str, Any]:
    """Field zone attempt frequencies only — NOT routes vs CB."""
    passes = df[df["play_type"] == "pass"]
    if "pass_location" not in passes.columns:

    out = {"by_team": {}, "label": "pass_location_NOT_route_or_coverage"}
    for team, g in passes.groupby("posteam"):
        vc = g["pass_location"].value_counts(dropna=False).to_dict()
        out["by_team"][team] = {str(k): int(v) for k, v in vc.items()}
    out["status"] = "OK_LOCATION_ONLY"
    return out

            "blitz", "was_pressure", "number_of_pass_rushers",
            "defense_man_zone_type", "defense_coverage_type", "route",

        "ns": {

            "rz_n": rz.get("n_total"),
            "qb_hit_n": qh.get("n_total"),
        },
    }

"""Bounded game state + explicit event transitions + period scoreboards."""
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Optional


def _empty_periods():
    return {"q1": [0, 0], "q2": [0, 0], "q3": [0, 0], "q4": [0, 0], "ot": [0, 0]}


@dataclass
class GameState:
    home: str
    away: str
    possession: Optional[str] = None
    qtr: int = 1
    half: int = 1
    game_seconds_remaining: int = 900
    down: int = 1
    ydstogo: int = 10
    yardline_100: int = 75
    score_home: int = 0
    score_away: int = 0
    in_ot: bool = False
    final: bool = False
    events: list = field(default_factory=list)
    player_counts: dict = field(default_factory=lambda: {"pass_yards": {}, "rush_yards": {}, "rec_yards": {}})
    sparse_flags: list = field(default_factory=list)
    reg_score_home: Optional[int] = None
    reg_score_away: Optional[int] = None
    period_scores: dict = field(default_factory=_empty_periods)

    @classmethod
    def new_game(cls, home: str, away: str) -> "GameState":
        return cls(home=home, away=away, period_scores=_empty_periods())

    def _period_key(self) -> str:
        if self.qtr <= 0:
            return "q1"
        if self.qtr == 1:
            return "q1"
        if self.qtr == 2:
            return "q2"
        if self.qtr == 3:
            return "q3"
        if self.qtr == 4:
            return "q4"
        return "ot"

    def add_score(self, team: str, pts: int) -> None:
        """Credit team total and the *current* quarter/OT bucket (boundary = current qtr)."""
        idx = 0 if team == self.home else 1
        if team == self.home:
            self.score_home += pts
        else:
            self.score_away += pts
        key = self._period_key()
        self.period_scores[key][idx] += pts
        self.events.append({"type": "score_credit", "team": team, "pts": pts, "period": key, "qtr": self.qtr})

    def period_snapshot(self) -> dict:
        ps = self.period_scores
        h1_h, h1_a = ps["q1"][0] + ps["q2"][0], ps["q1"][1] + ps["q2"][1]
        h2_h, h2_a = ps["q3"][0] + ps["q4"][0], ps["q3"][1] + ps["q4"][1]
        reg_h, reg_a = h1_h + h2_h, h1_a + h2_a
        return {
            "q1_home": ps["q1"][0], "q1_away": ps["q1"][1],
            "q2_home": ps["q2"][0], "q2_away": ps["q2"][1],
            "q3_home": ps["q3"][0], "q3_away": ps["q3"][1],
            "q4_home": ps["q4"][0], "q4_away": ps["q4"][1],
            "h1_home": h1_h, "h1_away": h1_a,
            "h2_home": h2_h, "h2_away": h2_a,
            "ot_home": ps["ot"][0], "ot_away": ps["ot"][1],
            "reg_home": reg_h, "reg_away": reg_a,
            "fg_home": self.score_home, "fg_away": self.score_away,
        }


def apply_kickoff(st: GameState, *, receiving: str, yardline_100: int = 75) -> GameState:
    st.possession = receiving
    st.yardline_100 = int(yardline_100)
    st.down = 1
    st.ydstogo = 10
    st.events.append({"type": "kickoff", "receiving": receiving, "yardline_100": yardline_100, "qtr": st.qtr})
    return st


def maybe_halftime(st: GameState, *, q3_receive: str) -> GameState:
    if st.qtr == 2 and st.game_seconds_remaining <= 0 and not st.final:
        st.events.append({"type": "halftime", "score": (st.score_home, st.score_away)})
        st.qtr = 3
        st.half = 2
        st.game_seconds_remaining = 900
        apply_kickoff(st, receiving=q3_receive, yardline_100=75)
    return st


def start_ot_if_tied(st: GameState) -> GameState:
    if st.qtr >= 4 and st.game_seconds_remaining <= 0 and not st.final:
        st.reg_score_home = st.score_home
        st.reg_score_away = st.score_away
        if st.score_home == st.score_away:
            st.in_ot = True
            st.qtr = 5
            st.game_seconds_remaining = 600
            st.events.append({"type": "ot_start", "rule": "single_possession_approx_declared_limitation"})
            apply_kickoff(st, receiving=st.away, yardline_100=75)
        else:
            st.final = True
            st.events.append({"type": "final", "score": (st.score_home, st.score_away)})
    return st

SCOPE_FOCUS = {"JAX", "BAL"}  # target slate focus; opponents allowed as historical partners


def list_jaxbal_games(pbp_path: str | Path) -> list[dict]:
    cols = ["game_id", "week", "home_team", "away_team", "play_id", "total_home_score", "total_away_score", "season"]
    t = pd.read_parquet(pbp_path, columns=cols)
    t = t[(t["season"] == 2026) & (t["week"] <= 8)]
    inv = t["home_team"].isin(SCOPE_FOCUS) | t["away_team"].isin(SCOPE_FOCUS)
    rows = []
    for gid, gg in t[inv].groupby("game_id"):
        home, away = gg["home_team"].iloc[0], gg["away_team"].iloc[0]
        last = gg.sort_values("play_id").iloc[-1]
        rows.append({
            "game_id": gid,
            "week": int(gg["week"].iloc[0]),
            "home": home,
            "away": away,
            "actual_home": float(last["total_home_score"]),
            "actual_away": float(last["total_away_score"]),
            "focus_teams": sorted(set([home, away]) & SCOPE_FOCUS),
        })
    return sorted(rows, key=lambda r: (r["week"], r["game_id"]))


def build_asof_history(pbp_path, *, teams, season, before_week):
    """Legacy helper: multi-team offense catalog, own-season weeks < before_week."""
    return build_asof_history_for_game(
        pbp_path, home=teams[0], away=teams[1] if len(teams) > 1 else teams[0],
        season=season, week=before_week,
    )


def build_asof_history_for_game(
    pbp_path: str | Path,
    *,
    home: str,
    away: str,
    season: int,
    week: int,
) -> dict:
    """Load BOTH teams' posteam plays for weeks strictly before `week`, same season only.

    No prior-year pooling. No league-wide silent fallback.
    """
    if week <= 1:
        return {
            "status": "EMPTY_ASOF",
            "df": None,
            "evidence": {
                "n_plays": 0, "n_plays_home": 0, "n_plays_away": 0,
                "reason": "NO_OWN_SEASON_HISTORY",
                "prior_year_pooled": False, "league_pooled": False,
            },
        }
    import pyarrow.parquet as pq
    path = Path(pbp_path)
    available = set(pq.ParquetFile(path).schema_arrow.names)
    use = [c for c in COLS if c in available]
    for extra in ("season", "qb_hit", "qb_scramble", "sack", "complete_pass"):
        if extra in available and extra not in use:
            use.append(extra)
    df = pd.read_parquet(path, columns=use)
    teams = [home, away]
    mask = (
        (df["season"] == season)
        & (df["week"] < week)
        & (df["week"] >= 1)
        & (df["posteam"].isin(teams))
    )
    off = df.loc[mask].copy()
    n_h = int((off["posteam"] == home).sum())
    n_a = int((off["posteam"] == away).sum())
    evid = {
        "n_plays": int(len(off)),
        "n_plays_home": n_h,
        "n_plays_away": n_a,
        "teams": teams,
        "weeks": [1, week - 1],
        "season": season,
        "prior_year_pooled": False,
        "league_pooled": False,
        "source": str(path),
    }
    if n_h == 0 or n_a == 0:
        return {"status": "SPARSE_ASOF_ONE_TEAM", "df": off.reset_index(drop=True) if len(off) else None, "evidence": evid}
    if off.empty:
        return {"status": "EMPTY_ASOF", "df": None, "evidence": evid}
    # Guard: no future weeks
    if off["week"].max() >= week:
        return {"status": "WEEK_FILTER_FAILED", "df": None, "evidence": evid}
    return {"status": "OK", "df": off.reset_index(drop=True), "evidence": evid}


def _asof_team_ppg(pbp_path, team: str, season: int, before_week: int) -> float | None:
    """Simple baseline: mean points scored in completed games before week."""
    cols = ["game_id", "week", "home_team", "away_team", "play_id", "total_home_score", "total_away_score", "season"]
    t = pd.read_parquet(pbp_path, columns=cols)
    t = t[(t["season"] == season) & (t["week"] < before_week) & (t["week"] >= 1)]
    inv = (t["home_team"] == team) | (t["away_team"] == team)
    scores = []
    for gid, gg in t[inv].groupby("game_id"):
        last = gg.sort_values("play_id").iloc[-1]
        scores.append(float(last["total_home_score"] if gg["home_team"].iloc[0] == team else last["total_away_score"]))
    return float(np.mean(scores)) if scores else None

_PERIOD_KEYS = [
    "q1_home", "q1_away", "q2_home", "q2_away", "q3_home", "q3_away", "q4_home", "q4_away",
    "h1_home", "h1_away", "h2_home", "h2_away", "reg_home", "reg_away", "fg_home", "fg_away",
]


def reconstruct_actual_periods(pbp_path: str | Path, game_id: str) -> dict[str, Any]:
    t = period_table_from_pbp(load_pbp(pbp_path))
    hit = t[t["game_id"] == game_id]
    if hit.empty:
        return {"status": "MISSING_GAME", "game_id": game_id}
    r = hit.iloc[0]
    out = {"status": "OK", "game_id": game_id}
    for k in _PERIOD_KEYS:
        out[k] = int(r[k]) if k in r.index and pd.notna(r[k]) else None
    # reconcile
    if out["h1_home"] is not None:
        out["recon_h1"] = (out["q1_home"] + out["q2_home"] == out["h1_home"]) and (
            out["q1_away"] + out["q2_away"] == out["h1_away"]
        )
        out["recon_reg"] = (out["h1_home"] + out["h2_home"] == out["reg_home"]) and (
            out["h1_away"] + out["h2_away"] == out["reg_away"]
        )
    return out


def _brier_home_win(pred_h: np.ndarray, pred_a: np.ndarray, actual_h: float, actual_a: float) -> float:
    """Brier for P(home period points > away); ties → outcome 0 (non-win), pred = P(h>a)."""
    p = float(np.mean(pred_h > pred_a))
    y = 1.0 if actual_h > actual_a else 0.0
    return (p - y) ** 2


def grade_periods_for_games(
    pbp_path: str | Path,
    grade_rows: list[dict],
    *,
    n_sims: int = 150,
    seed_base: int = 20261105,
    config_path: str | Path | None = None,
) -> dict[str, Any]:
    """For each GRADED_FULL row, resim with period snapshots; compare to reconstructed actuals."""
    root = Path(__file__).resolve().parents[1]
    cfg_path = Path(config_path) if config_path else root / "config/jax_bal_tnf_2026-11-05_data_only.json"
    base_cfg = load_config(cfg_path)
    by_game = []
    # collectors for metrics
    err = {k: [] for k in [
        "q1_home", "q1_away", "h1_home", "h1_away", "h2_home", "h2_away",
        "reg_home", "reg_away", "fg_home", "fg_away",
    ]}
    brier = {"h1": [], "h2": [], "reg": [], "fg": []}

    for g in grade_rows:
        if g.get("grade_status") != "GRADED_FULL":
            by_game.append({**g, "period_grade_status": "SKIPPED_NOT_GRADED_FULL"})
            continue
        act = reconstruct_actual_periods(pbp_path, g["game_id"])
        if act["status"] != "OK":
            by_game.append({**g, "period_grade_status": "ACTUAL_PERIODS_MISSING", "actual_periods": act})
            continue
        asof = build_asof_history_for_game(
            pbp_path, home=g["home"], away=g["away"], season=2026, week=g["week"]
        )
        if asof["status"] != "OK":
            by_game.append({**g, "period_grade_status": f"ASOF_{asof['status']}"})
            continue
        cfg = dict(base_cfg)

        cat = PlayCatalog(asof["df"], min_n=int(cfg["min_n_state"]))
        seed = stable_seed(seed_base, g["week"], g["game_id"] + "|period")

        snaps = []
        for _ in range(n_sims):

            snap = st.period_snapshot()

                snap["reg_home"] = st.reg_score_home
                snap["reg_away"] = st.reg_score_away
            snaps.append(snap)
        pred = {k: float(np.mean([s[k] for s in snaps])) for k in err}
        # arrays for brier
        arr = {k: np.asarray([s[k] for s in snaps]) for k in ("h1_home", "h1_away", "h2_home", "h2_away", "reg_home", "reg_away", "fg_home", "fg_away")}
        row_err = {}
        for k in err:
            row_err[f"pred_{k}"] = pred[k]
            row_err[f"actual_{k}"] = act[k]
            row_err[f"abs_err_{k}"] = abs(pred[k] - act[k]) if act[k] is not None else None
            if row_err[f"abs_err_{k}"] is not None:
                err[k].append(row_err[f"abs_err_{k}"])
        row_brier = {
            "brier_h1_home_win": _brier_home_win(arr["h1_home"], arr["h1_away"], act["h1_home"], act["h1_away"]),
            "brier_h2_home_win": _brier_home_win(arr["h2_home"], arr["h2_away"], act["h2_home"], act["h2_away"]),
            "brier_reg_home_win": _brier_home_win(arr["reg_home"], arr["reg_away"], act["reg_home"], act["reg_away"]),
            "brier_fg_home_win": _brier_home_win(arr["fg_home"], arr["fg_away"], act["fg_home"], act["fg_away"]),
        }
        brier["h1"].append(row_brier["brier_h1_home_win"])
        brier["h2"].append(row_brier["brier_h2_home_win"])
        brier["reg"].append(row_brier["brier_reg_home_win"])
        brier["fg"].append(row_brier["brier_fg_home_win"])
        by_game.append({
            **{k: g[k] for k in ("game_id", "week", "home", "away", "grade_status")},
            "period_grade_status": "GRADED_PERIODS",
            "n_sims": n_sims,
            "seed": seed,
            "actual_recon_h1": act.get("recon_h1"),
            "actual_recon_reg": act.get("recon_reg"),

    metrics = {

        "mae_q1_home": float(np.mean(err["q1_home"])) if err["q1_home"] else None,

        "mae_h1_away": float(np.mean(err["h1_away"])) if err["h1_away"] else None,
        "mae_h2_home": float(np.mean(err["h2_home"])) if err["h2_home"] else None,
        "mae_h2_away": float(np.mean(err["h2_away"])) if err["h2_away"] else None,
        "mae_reg_home": float(np.mean(err["reg_home"])) if err["reg_home"] else None,

"""Drive-length / plays-per-game realism vs team W1–8 actuals (development diagnostic)."""
from __future__ import annotations
from pathlib import Path
from typing import Any
import numpy as np
import pandas as pd

def _actual_pace(inv: pd.DataFrame, teams=("JAX", "BAL")) -> dict:
    # offense snaps per team-game
    snaps = inv[inv["play_type"].isin(["pass", "run", "qb_kneel", "qb_spike"])]
    snaps = snaps[snaps["posteam"].isin(teams)]
    per = snaps.groupby(["posteam", "game_id"]).size()
    # drives: unique game_id+drive where posteam in teams
    drives = inv[inv["posteam"].isin(teams) & inv["drive"].notna()]
    drive_lens = drives.groupby(["posteam", "game_id", "drive"]).size()
    return {
        "mean_offense_snaps_per_team_game": float(per.mean()) if len(per) else None,
        "median_offense_snaps_per_team_game": float(per.median()) if len(per) else None,
        "n_team_games": int(per.shape[0]),
        "drive_length_plays": {
            "mean": float(drive_lens.mean()) if len(drive_lens) else None,
            "median": float(drive_lens.median()) if len(drive_lens) else None,
            "n_drives": int(len(drive_lens)),
        },
        "by_team_mean_snaps": {t: float(per[t].mean()) if t in per.index.get_level_values(0) else None for t in teams},
    }


def compare_sim_vs_actual_pace(cfg: dict, *, pbp_path: str | Path, n_sims: int = 100, seed: int = 3) -> dict[str, Any]:
    hist = load_team_history(
        pbp_path,
        teams=cfg["history"]["teams"],
        season=cfg["history"]["season"],
        weeks=(cfg["history"]["weeks_inclusive"][0], cfg["history"]["weeks_inclusive"][-1]),
    )
    inv = load_involvement_plays(
        pbp_path,
        teams=cfg["history"]["teams"],
        season=cfg["history"]["season"],
        weeks=(cfg["history"]["weeks_inclusive"][0], cfg["history"]["weeks_inclusive"][-1]),
    )
    actual = _actual_pace(inv, teams=tuple(cfg["history"]["teams"]))
    df = hist["df"]
    if cfg.get("adjustments", {}).get("enabled"):
        df, _ = apply_forward_qb_filter(df, cfg["adjustments"])
    cat = PlayCatalog(df, min_n=int(cfg["min_n_state"]))
    rng = np.random.default_rng(seed)
    snap_counts = []
    punt_counts = []
    td_counts = []
    # approximate drive count = kickoffs + turnovers + punts received... use punt+ko as possession proxy
    poss_proxy = []
    for _ in range(n_sims):
        st = simulate_one(df, cat, cfg, np.random.default_rng(int(rng.integers(0, 2**31 - 1))))
        n_snap = sum(1 for e in st.events if e.get("type") == "snap")
        n_punt = sum(1 for e in st.events if e.get("type") == "punt")
        n_td = sum(1 for e in st.events if e.get("type") == "touchdown")
        n_ko = sum(1 for e in st.events if e.get("type") == "kickoff")
        snap_counts.append(n_snap)
        punt_counts.append(n_punt)
        td_counts.append(n_td)
        poss_proxy.append(n_ko + n_punt)  # rough possessions
    sim = {
        "mean_snaps_per_game": float(np.mean(snap_counts)),
        "median_snaps_per_game": float(np.median(snap_counts)),
        "mean_punts": float(np.mean(punt_counts)),
        "mean_tds": float(np.mean(td_counts)),
        "mean_possession_proxy": float(np.mean(poss_proxy)),
        "n_sims": n_sims,
    }
    # Actual team-game snaps are per ONE offense; sim snaps are BOTH offenses combined
    actual_both = None
    if actual["mean_offense_snaps_per_team_game"] is not None:
        actual_both = 2.0 * actual["mean_offense_snaps_per_team_game"]
    return {
        "label": "development_in_sample_diagnostic_not_accuracy",
        "actual": actual,
        "actual_both_offenses_snaps_approx": actual_both,
        "sim": sim,
        "comparison": {
            "sim_snaps_minus_2x_team_snaps": (
                sim["mean_snaps_per_game"] - actual_both if actual_both is not None else None
            ),
            "note": "Sim snap count is joint (both teams); actual mean is per-team offense — compare to 2× actual",
        },
    }

"""Apply ONLY supported effects to the snap catalog — must change joint sampled events.

Supported:
  - annotate snaps with explosive flag (definition-based)
  - annotate pass snaps with is_blitz_ftn when FTN joined
  - optional script_adjusted: down-weight (not invent coeffs) by excluding extreme
    historical explosive long TDs? NO — don't invent.

Actual sim effect that is testable without invented coefficients:
  - When effects_enabled, pass-play sampling can stratify: with probability p_blitz
    sample from blitz-joined rows when available for that team; else fall back with flag.
  - p_blitz estimated empirically as team's faced/own blitz rate from FTN join — data-driven rate, not hand weight.
"""
from __future__ import annotations
from typing import Any, Optional
import numpy as np
import pandas as pd
from effects.definitions import EXPLOSIVE_PASS_YARDS, EXPLOSIVE_RUSH_YARDS


def annotate_snaps(df: pd.DataFrame) -> pd.DataFrame:
    out = df.copy()
    yg = pd.to_numeric(out.get("yards_gained"), errors="coerce").fillna(0)
    out["is_explosive"] = (
        ((out["play_type"] == "pass") & (yg >= EXPLOSIVE_PASS_YARDS))
        | ((out["play_type"] == "run") & (yg >= EXPLOSIVE_RUSH_YARDS))
    )
    if "qb_hit" in out.columns:
        out["pressure_proxy_qb_hit"] = out["qb_hit"].fillna(0).astype(int) == 1
    else:
        out["pressure_proxy_qb_hit"] = False
    if "is_blitz_ftn" not in out.columns:
        out["is_blitz_ftn"] = False
        out["ftn_joined"] = False
    return out


class EffectsCatalog:
    """Wraps PlayCatalog-like sampling with optional blitz stratification (empirical rate)."""

    def __init__(self, base_catalog, annotated: pd.DataFrame, *, enable_blitz_strata: bool):
        self.base = base_catalog
        self.df = annotated
        self.enable_blitz_strata = enable_blitz_strata
        self.min_n = base_catalog.min_n
        self.specials = base_catalog.specials
        self.fourth_probs = base_catalog.fourth_probs
        self.cells = base_catalog.cells
        self.by_down = base_catalog.by_down
        self.by_team = base_catalog.by_team
        # empirical blitz rate by posteam among joined passes
        self.blitz_rate = {}
        passes = annotated[(annotated["play_type"] == "pass") & (annotated.get("ftn_joined", False) == True)]
        for team, g in passes.groupby("posteam"):
            self.blitz_rate[team] = float(g["is_blitz_ftn"].mean()) if len(g) else 0.0
        self.defense_blitz_rate = {}
        self.blitz_pools = {}
        self.noblitz_pools = {}
        for team, g in annotated[annotated["play_type"] == "pass"].groupby("posteam"):
            bj = g[g["is_blitz_ftn"] & g["ftn_joined"]]
            nj = g[(~g["is_blitz_ftn"]) & g["ftn_joined"]]
            if len(bj) >= 5:
                self.blitz_pools[team] = bj
            if len(nj) >= 5:
                self.noblitz_pools[team] = nj

    def set_defense_blitz_rates(self, rates: dict):
        """Empirical FTN blitz rates allowed by defense team (data-driven, not hand weights)."""
        self.defense_blitz_rate = dict(rates or {})

    def sample_snap(self, rng: np.random.Generator, st, allow_pool: bool):
        if (
            self.enable_blitz_strata
            and st.down <= 3
            and st.possession in self.blitz_pools
            and st.possession in self.noblitz_pools
        ):
            # Prefer opponent defense empirical blitz rate when present; else offense-faced rate
            defense = st.home if st.possession == st.away else st.away
            if defense in self.defense_blitz_rate:
                p = self.defense_blitz_rate[defense]
                tag_prefix = "EFFECT_DEF_BLITZ"
            elif st.possession in self.blitz_rate:
                p = self.blitz_rate[st.possession]
                tag_prefix = "EFFECT_BLITZ"
            else:
                return self.base.sample_snap(rng, st, allow_pool)
            use_blitz = bool(rng.random() < p)
            pool = self.blitz_pools[st.possession] if use_blitz else self.noblitz_pools[st.possession]
            row = pool.iloc[int(rng.integers(0, len(pool)))]
            tag = (tag_prefix + "_STRATA") if use_blitz else (tag_prefix + "_NO_STRATA")
            return row, tag
        return self.base.sample_snap(rng, st, allow_pool)

def _find_ftn() -> Optional[Path]:
    for p in FTN_CANDIDATES:
        if p.exists():
            return p
    return None


def load_ftn_join(pbp: pd.DataFrame, *, teams=("JAX", "BAL"), weeks=(1, 8)) -> dict[str, Any]:
    ftn_path = _find_ftn()
    meta = {
        "ftn_path": str(ftn_path) if ftn_path else None,
        "status": "MISSING_FTN" if ftn_path is None else "OK",
        "man_zone_mofo_mofc_in_ftn": False,
        "note": "FTN 2026 local extract has n_blitzers/n_pass_rushers; no man/zone/MOFO/MOFC/route columns",
    }
    if ftn_path is None:
        return {"meta": meta, "joined": None}
    ftn = pd.read_parquet(ftn_path)
    # join keys
    left = pbp.copy()
    if "game_id" not in left.columns or "play_id" not in left.columns:
        meta["status"] = "PBP_MISSING_KEYS"
        return {"meta": meta, "joined": None}
    ftn = ftn.rename(columns={"nflverse_game_id": "game_id", "nflverse_play_id": "play_id"})
    # play_id types
    left["play_id"] = pd.to_numeric(left["play_id"], errors="coerce")
    ftn["play_id"] = pd.to_numeric(ftn["play_id"], errors="coerce")
    j = left.merge(ftn[["game_id", "play_id", "n_blitzers", "n_pass_rushers", "date_pulled"]], on=["game_id", "play_id"], how="left")
    j["is_blitz_ftn"] = j["n_blitzers"].fillna(0) >= 1
    j["ftn_joined"] = j["n_blitzers"].notna()
    meta["n_pbp"] = int(len(left))
    meta["n_joined"] = int(j["ftn_joined"].sum())
    meta["join_rate"] = float(j["ftn_joined"].mean())
    meta["date_pulled_sample"] = str(j["date_pulled"].dropna().iloc[0]) if j["date_pulled"].notna().any() else None
    return {"meta": meta, "joined": j}


def explosive_tables(df: pd.DataFrame) -> dict[str, Any]:
    """Offensive explosive rates + defensive explosives allowed (when team is defteam)."""
    out = {
        "definitions": {
            "explosive_pass_yards": EXPLOSIVE_PASS_YARDS,
            "explosive_rush_yards": EXPLOSIVE_RUSH_YARDS,
        },
        "offense": {},
        "defense_allowed": {},
    }
    for team, g in df.groupby("posteam"):
        passes = g[g["play_type"] == "pass"]
        runs = g[g["play_type"] == "run"]
        exp_p = passes[pd.to_numeric(passes["yards_gained"], errors="coerce").fillna(0) >= EXPLOSIVE_PASS_YARDS]
        exp_r = runs[pd.to_numeric(runs["yards_gained"], errors="coerce").fillna(0) >= EXPLOSIVE_RUSH_YARDS]
        out["offense"][team] = {
            "pass_attempts": int(len(passes)),
            "explosive_passes": int(len(exp_p)),
            "p_explosive_pass": float(len(exp_p) / len(passes)) if len(passes) else None,
            "rush_attempts": int(len(runs)),
            "explosive_rushes": int(len(exp_r)),
            "p_explosive_rush": float(len(exp_r) / len(runs)) if len(runs) else None,
            "status": "OK" if len(passes) + len(runs) >= 30 else "SPARSE",
        }
    # defense allowed: need defteam on rows — offense df is posteam-filtered; use games via full involvements if present
    if "defteam" in df.columns:
        for team, g in df.groupby("defteam"):
            if team not in ("JAX", "BAL"):
                continue
            passes = g[g["play_type"] == "pass"]
            runs = g[g["play_type"] == "run"]
            exp_p = passes[pd.to_numeric(passes["yards_gained"], errors="coerce").fillna(0) >= EXPLOSIVE_PASS_YARDS]
            exp_r = runs[pd.to_numeric(runs["yards_gained"], errors="coerce").fillna(0) >= EXPLOSIVE_RUSH_YARDS]
            out["defense_allowed"][team] = {
                "pass_attempts_faced": int(len(passes)),
                "explosive_passes_allowed": int(len(exp_p)),
                "p_exp_pass_allowed": float(len(exp_p) / len(passes)) if len(passes) else None,
                "rush_attempts_faced": int(len(runs)),
                "explosive_rushes_allowed": int(len(exp_r)),
                "p_exp_rush_allowed": float(len(exp_r) / len(runs)) if len(runs) else None,
                "status": "OK" if len(passes) + len(runs) >= 30 else "SPARSE",
                "note": "Only plays where this team was defteam in team-filtered rows; if offense-only filter, use games_df",
            }
    return out


def qb_completion_splits(joined: pd.DataFrame) -> dict[str, Any]:
    """Overall + qb_hit PROXY + FTN blitz splits. Never label qb_hit as measured pressure."""
    passes = joined[joined["play_type"] == "pass"].copy()
    out = {"by_passer": {}, "labels": {
        "qb_hit": "PROXY_NOT_MEASURED_PRESSURE",
        "blitz": "FTN_n_blitzers_ge_1_when_joined",
        "true_pressure_charted": "MISSING_in_local_FTN_extract",
    }}
    if passes.empty:
        return out
    for (team, qb), g in passes.groupby(["posteam", "passer_player_name"]):
        if not isinstance(qb, str):
            continue
        def comp(x):
            if len(x) == 0 or "complete_pass" not in x.columns:
                return None, 0
            return float((x["complete_pass"] == 1).mean()), int(len(x))
        c_all, n_all = comp(g)
        hit = g[g.get("qb_hit", 0) == 1] if "qb_hit" in g.columns else g.iloc[0:0]
        nohit = g[g.get("qb_hit", 0) != 1] if "qb_hit" in g.columns else g
        c_hit, n_hit = comp(hit)
        c_nh, n_nh = comp(nohit)
        if "is_blitz_ftn" in g.columns and g["ftn_joined"].any():
            bl = g[g["is_blitz_ftn"] & g["ftn_joined"]]
            nbl = g[(~g["is_blitz_ftn"]) & g["ftn_joined"]]
            c_bl, n_bl = comp(bl)
            c_nbl, n_nbl = comp(nbl)
        else:
            c_bl = n_bl = c_nbl = n_nbl = None
        out["by_passer"][f"{team}:{qb}"] = {
            "n_att": n_all, "comp_pct": c_all,
            "n_qb_hit_proxy": n_hit, "comp_pct_qb_hit_proxy": c_hit,
            "n_no_qb_hit": n_nh, "comp_pct_no_qb_hit": c_nh,
            "n_blitz_ftn": n_bl, "comp_pct_blitz_ftn": c_bl,
            "n_no_blitz_ftn": n_nbl, "comp_pct_no_blitz_ftn": c_nbl,
            "status": "OK" if n_all >= 20 else "SPARSE",
        }
    return out


def recent_form_miss_audit(df: pd.DataFrame, *, last_n_games: int = 3) -> dict[str, Any]:
    """Scenario-first recent form: last N games PPG / explosive — audit only, no auto cut."""
    out = {"label": "SCENARIO_FIRST_MISS_AUDIT_NO_AUTO_CUT", "by_team": {}}
    if "game_id" not in df.columns:
        return out
    for team in ("JAX", "BAL"):
        # approximate: unique games where posteam==team ordered by week
        sub = df[df["posteam"] == team]
        if sub.empty:
            continue
        games = sub.groupby("game_id").agg(week=("week", "first"), plays=("play_id", "count")).sort_values("week")
        recent = games.tail(last_n_games)
        out["by_team"][team] = {
            "recent_game_ids": list(map(str, recent.index)),
            "n_games": int(len(recent)),
            "note": "Not used as automatic veto; scenario cross-check only",
        }
    return out
